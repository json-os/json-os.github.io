Developer Experience

Elegant by design

app.js 
import os from 'jsonos';

// Fetch tasks from Alice's pod
const tasks = await os.read('https://alice.solidweb.org/tasks.json');

// Render them
os.render(tasks);

// Make a change
tasks.push({ title: 'Learn jsonos', done: false });

// Save to your pod
await os.write('/tasks.json', tasks);
